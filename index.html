One-time Firebase setup (Firestore rules)

Your current rules are kept as they are. Add this block (already included in the ready-made firestore.rules file) inside match /databases/{database}/documents { … } after the certDb block, then Publish. It uses your existing isAdmin() / isMarksStaff() (roles in users/{uid}):

// ================= ADMISSION (promotion, new admission, online form) — SEPARATE RULE =================
    // Public online form (admission.html) can only CREATE a "pending" application, no login needed.
    // Staff (staff/teacher) verify; only admin approves; admit needs admin approval to exist first.
    match /admApps/{id} {
      allow create: if request.resource.data.status == 'pending'
        && request.resource.data.name is string && request.resource.data.name.size() < 120
        && request.resource.data.keys().size() < 90
        && request.resource.data.get('photo', '').size() < 150000
        && !('staffOk' in request.resource.data) && !('adminOk' in request.resource.data);
      allow read: if isAdmin() || isMarksStaff();
      allow update: if isAdmin()
        || (isMarksStaff()
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasOnly(['staffOk', 'staffNote', 'status', 'rejectReason', 'admitted'])
            && request.resource.data.status != 'approved'
            && (request.resource.data.status != 'admitted' || 'adminOk' in resource.data));
      allow delete: if isAdmin();
    }
    // Placed students (promoted + new) and roll-number counters, per admission year
    // NEW admissions (type 'new', the admission register) can be written by ADMIN only; staff still place promoted students.
    match /admStudents/{id} {
      allow read: if isAdmin() || isMarksStaff();
      allow create: if isAdmin() || (isMarksStaff() && request.resource.data.type != 'new');
      allow update: if isAdmin() || (isMarksStaff() && resource.data.type != 'new' && request.resource.data.type != 'new');
      allow delete: if isAdmin();
    }
    // REG__ counters (admission register serial numbers) are admin-only
    match /admCounters/{id} {
      allow read: if isAdmin() || isMarksStaff();
      allow create, update: if isAdmin() || (isMarksStaff() && !id.matches('REG_.*'));
      allow delete: if isAdmin();
    }
    // =================================================================================================

Roles: admin approves and can delete; staff / teacher verify, admit and place students. “Publish to Data & Year” writes certDb, which your rules allow for admin only.

One-time Firebase setup (Firestore rules) — required for this screen

Add / replace these blocks in your Firestore rules, then Publish. Without them class teachers cannot register or save, and the “locked” state is only enforced on screen.

// ===== PROMOTION / CLASS-TEACHER ACCESS =====
// Put these inside:  match /databases/{database}/documents { ... }
// Uses your existing isAdmin() and isMarksStaff(). New role value in users/{uid}.role = 'classteacher'
// (kept separate from 'teacher' so class teachers do NOT get your other marks/portal permissions).

function uDoc() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
function isCT() {
  return request.auth != null
    && exists(/databases/$(database)/documents/users/$(request.auth.uid))
    && uDoc().role == 'classteacher'
    && uDoc().get('disabled', false) == false;
}
function ctOwns(c, s) { return isCT() && uDoc().cls == c && uDoc().sec == s; }
function sheetKey(d) { return string(d.year) + '_' + d.cls + '_' + (d.sec == '' ? '-' : d.sec); }
function sheetOf(d) { return get(/databases/$(database)/documents/admAlloc/$(sheetKey(d))).data; }
function ctPlaces(d) {
  return ctOwns(d.cls, d.sec) && d.type == 'promoted'
    && d.get('allocKey', '') == sheetKey(d) && sheetOf(d).status == 'draft';
}

// 1) Invite links created by admin; the public join page can read ONE invite by its secret code.
match /teacherInvites/{code} {
  allow get: if true;
  allow list: if isAdmin();
  allow create, delete: if isAdmin();
  allow update: if isAdmin()
    || (request.auth != null && resource.data.status == 'open'
        && request.resource.data.status == 'used'
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'uid', 'email', 'usedAt']));
}

// 2) A teacher may create ONLY their own users/{uid} document, and only for an open invite.
//    (This is added to your existing users rule - Firestore ORs matching rules.)
match /users/{uid} {
  allow read: if request.auth != null && request.auth.uid == uid;
  allow create: if request.auth != null && request.auth.uid == uid
    && request.resource.data.keys().hasOnly(['role', 'name', 'email', 'mobile', 'cls', 'sec', 'year', 'inviteCode'])
    && request.resource.data.role == 'classteacher'
    && get(/databases/$(database)/documents/teacherInvites/$(request.resource.data.inviteCode)).data.status == 'open'
    && get(/databases/$(database)/documents/teacherInvites/$(request.resource.data.inviteCode)).data.cls == request.resource.data.cls
    && get(/databases/$(database)/documents/teacherInvites/$(request.resource.data.inviteCode)).data.sec == request.resource.data.sec
    && get(/databases/$(database)/documents/teacherInvites/$(request.resource.data.inviteCode)).data.year == request.resource.data.year;
}

// 3) Allocation sheets (draft -> final). A class teacher can only touch THEIR class-section and
//    only while it is a draft. Locked (final) sheets can be changed by admin only.
match /admAlloc/{id} {
  allow read: if isAdmin() || isCT();
  allow create: if isAdmin()
    || (ctOwns(request.resource.data.cls, request.resource.data.sec)
        && request.resource.data.status == 'draft' && id == sheetKey(request.resource.data));
  allow update: if isAdmin()
    || (ctOwns(resource.data.cls, resource.data.sec) && resource.data.status == 'draft'
        && request.resource.data.cls == resource.data.cls && request.resource.data.sec == resource.data.sec
        && request.resource.data.status in ['draft', 'final']);
  allow delete: if isAdmin();
}

// 4) REPLACE your existing admStudents and admCounters blocks with these two
//    (staff can no longer write promoted students directly - promotion goes through the sheets).
match /admStudents/{id} {
  allow read: if isAdmin() || isMarksStaff() || isCT();
  allow create: if isAdmin() || ctPlaces(request.resource.data);
  allow update: if isAdmin()
    || (ctPlaces(request.resource.data) && resource.data.get('allocKey', '') == request.resource.data.allocKey);
  allow delete: if isAdmin()
    || (isCT() && ctOwns(resource.data.cls, resource.data.sec)
        && resource.data.get('allocKey', '') != '' && sheetOf(resource.data).status == 'draft');
}
match /admCounters/{id} {
  allow read: if isAdmin() || isMarksStaff() || isCT();
  allow create, update: if isAdmin()
    || (isCT() && !id.matches('REG_.*') && ctOwns(request.resource.data.cls, request.resource.data.sec));
  allow delete: if isAdmin();
}

// 5) In your existing certDb rule, change the read line to:   allow read: if isAdmin() || isMarksStaff() || isCT();
//    so class teachers can load the student database for certificates and reports.
